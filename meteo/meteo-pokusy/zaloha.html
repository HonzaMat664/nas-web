Cílem je mít jedním příkazem zazálohované:

  • konfigurace systému (systemd služby, cron)

  • Python projekt + virtuální prostředí

  • CSV data

  • Git repozitář (stav + remote)


struktura záloh:

/backups/
├─ meteo-backup-YYYY-MM-DD.tar.gz
└─ meteo-backup-latest.tar.gz



backup.sh:

#!/bin/bash
set -e

BACKUP_DIR="$HOME/backups"
DATE=$(date +%F)
NAME="meteo-backup-$DATE.tar.gz"
LATEST="meteo-backup-latest.tar.gz"

mkdir -p "$BACKUP_DIR"

sudo tar czf "$BACKUP_DIR/$NAME" \
/etc/systemd/system/meteostanice.service \
/etc/systemd/system/meteostanice.timer \
/etc/cron.d \
/home/honza/nas-web \
/home/honza/venv \
/var/www/html/meteostanice \
/etc/timezone \
/etc/localtime

cp "$BACKUP_DIR/$NAME" "$BACKUP_DIR/$LATEST"

echo "✅ Záloha hotová: $BACKUP_DIR/$NAME"

restore.sh:

#!/bin/bash
set -e

ARCHIVE=$1

if [ -z "$ARCHIVE" ]; then
echo "Použití: ./restore.sh meteo-backup-YYYY-MM-DD.tar.gz"
exit 1
fi

sudo tar xzf "$ARCHIVE" -C /

sudo systemctl daemon-reexec
sudo systemctl daemon-reload
sudo systemctl enable meteostanice.service
sudo systemctl start meteostanice.service

cd /home/honza/nas-web
source /home/honza/venv/bin/activate
pip install -r requirements.txt

echo "♻️ Obnova dokončena"

Ověření:

tar tzf ~/backups/meteo-backup-latest.tar.gz | less
